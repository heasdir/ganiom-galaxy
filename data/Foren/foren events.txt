# ============================================================
# EVENT 1: HISAIR SHIPYARD DESTROYED
# ============================================================

event "hisair_shipyard_destroyed"
	# Remove Hisair fleets from Nisar
	remove fleet "Hisair High Fleet"
	# Reduce Hisair presence in surrounding systems
	system "Nisar"
		add fleet "Hisair High Fleet" 0
		government "Uninhabited"
	# Foren gains influence
	system "Nisar"
		add fleet "Foren Defense Fleet" 80
		government "The Forens"
	# News
	news "Hisair Shipyard Destroyed"
		location
			government "The Forens"
			government "Gainen Republic"
			government "Reisbruth"
		name
			word "Hisair Shipyard Destroyed"
		message
			word
				"The Foren-Gainen coalition has destroyed the secret Hisair shipyard in Nisar."
				"Hisair's war effort is crippled. The galaxy rejoices."
				"Hisair's shipyard in Nisar is now a graveyard. Good riddance."

# EVENT 4: HISAIR DEFEATED — HISAR FALLS
# ============================================================

	# === HISAR — THE HEART OF HISAIR ===
	system "Hisar"
		remove fleet "Hisair High Fleet"
		remove fleet "Hisair Private Forces"
		government "The Forens"
		add fleet "Foren Occupation Fleet" 200
	planet "Hiswren"

		spaceport `"The Foren flag flies over Hiswren. Hisair is no more."`

		bribe 0

		add fleet "Gainen Defense Fleet" 100

		remove fleet "Hisair Private Forces"
		government "Gainen Republic"
		add fleet "Gainen Defense Fleet" 60
